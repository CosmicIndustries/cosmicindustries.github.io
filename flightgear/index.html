#!/bin/bash
# FFB Setup — Rock 5B+ pidff-6 kernel

# 1. Verify kernel
uname -r
# Must be: 6.1.84-8-rk2410-pidff-6

# 2. Load joystick module
sudo modprobe joydev
ls /dev/input/js*       # Should show js0
ls /dev/input/event*    # Find event node

# 3. Install tools
sudo apt install -y joystick evtest python3-evdev

# 4. Quick FFB test
fftest /dev/input/event0

# 5. Check FF capabilities
python3 - <<'EOF'
import evdev
d = evdev.InputDevice('/dev/input/event0')
caps = d.capabilities(verbose=True)
ff = caps.get(('EV_FF', 21), [])
print(f"Device: {d.name}")
for c in ff: print(' -', c)
EOF

# 6. Fix extlinux after any dpkg -i linux-image-*
sudo sed -i 's/^default l0/default l3/' \
  /boot/extlinux/extlinux.conf

Step 1 · Verify

      
Confirm /dev/input/js0

      
After booting pidff-6 kernel (l3), joydev must load before js0 appears.

      
sudo modprobe joydev
ls /dev/input/js*
# Expected: /dev/input/js0  (Evo Force)
# If missing: dmesg | grep -E "hid|input|joy"
uname -r  # Must show: 6.1.84-8-rk2410-pidff-6

    

    

      
Step 2 · Calibrate

      
jstest Axis Check

      
sudo apt install -y joystick
jstest /dev/input/js0
# Move all axes — verify roll, pitch, throttle, twist
# Note deadzone near center on axes 0 and 1
# Ctrl+C when done

# For calibration:
jscal /dev/input/js0

    

    

      
Step 3 · FG Config

      
Joystick XML Location

      
FlightGear looks in ~/.fgfs/Input/Joysticks/ first, then system paths. Use FFB tab to generate the full XML.

      
mkdir -p ~/.fgfs/Input/Joysticks/
# File: ~/.fgfs/Input/Joysticks/Saitek-Cyborg-Evo-Force.xml
# Name must match: fgfs --show-joystick output
fgfs --show-joystick 2>&1 | grep -i saitek

    

    

      
Step 4 · FFB Effects

      
Verify Force Feedback

      
sudo apt install -y evtest
# Find event node (Evo Force is usually event0):
evtest --query /dev/input/event0 EV_FF 0
# Exit code 10 = FF_RUMBLE supported ✓

# Quick rumble test:
python3 -c "
import evdev; from evdev import ecodes, ff
d = evdev.InputDevice('/dev/input/event0')
r = ff.Rumble(strong_magnitude=0x7FFF, weak_magnitude=0x7FFF)
e = ff.Effect(ecodes.FF_RUMBLE,-1,0,ff.Trigger(0,0),ff.Replay(500,0),ff.EffectType(ff_rumble_effect=r))
eid = d.upload_effect(e); d.write(ecodes.EV_FF,eid,1)
import time; time.sleep(0.6); d.erase_effect(eid)
print('FFB OK')
"

    

    

      
Step 5 · Throttle

      
Saitek Gold as Throttle

      
Plug Gold quadrant into second USB port. After modprobe joydev it appears as js1. Bind its axis 0 to throttle in FG XML or via FG's joystick dialog (Equipment → Joysticks).

      
ls /dev/input/js*  # js0=Evo, js1=Gold
jstest /dev/input/js1  # Verify throttle axis moves

    

    

      CONFIG_HID_PID=y — compiled into kernel, NOT a module. modprobe hid-pidff will fail with "not found" — this is correct. The driver loads automatically on USB plug-in.

Bulk Install

      
Full Fleet via SVN Loop

      
#!/bin/bash
# Run from ~ to install full fleet
BASE="https://svn.code.sf.net/p/flightgear/fgaddon/trunk/Aircraft"
DEST="$HOME/.fgfs/Aircraft"
mkdir -p "$DEST"

for ac in PA28R-200 PA28-181 PA28R-201 c172r c172p \
          AT-802 ag-cat bo105 Bonanza-V35B PA34-200T C208B; do
  echo "=== Installing $ac ==="
  svn checkout "$BASE/$ac" "$DEST/$ac" 2>&1 | tail -3
done
echo "Done."

CPU Affinity

      
Pin to A76 Big Cores

      
RK3588 big.LITTLE: CPUs 4-7 are A76 (2.4GHz big cores). Pin FG to them for best single-thread perf.

      
# Set performance governor first:
for cpu in 4 5 6 7; do
  echo performance | sudo tee \
    /sys/devices/system/cpu/cpu${cpu}/cpufreq/scaling_governor
done

# Launch with affinity:
taskset -c 4-7 /usr/local/bin/fgfs \
  --aircraft=PA28-161-160 --airport=KMCN \
  [... other flags ...]

Aircraft Auto-detect

      
Dynamic Selector Script

      
#!/bin/bash
# Auto-detect installed aircraft and present menu
AC_DIR="$HOME/.fgfs/Aircraft"
ACS=$(ls -d "$AC_DIR"/*/  2>/dev/null | xargs -I{} basename {})
CHOICE=$(echo "$ACS" | fzf --prompt="Aircraft> " --height=15)
[ -z "$CHOICE" ] && exit 0
/usr/local/bin/fgfs --aircraft="$CHOICE" \
  --airport=${FG_AIRPORT:-KMCN} \
  $(cat ~/.fgfs/fgfsrc | tr '\n' ' ')
